3. p2-Wiederherstellungspunkte vergleichen

Die Ausgabe danach bitte komplett an mich schicken.

$profileDir = "$env:USERPROFILE\.p2\org.eclipse.equinox.p2.engine\profileRegistry\C__Users_X603357_eclipse-abap-bsh-2026-06_eclipse-abap-bsh.profile"

Get-ChildItem -LiteralPath $profileDir -Filter "*.profile.gz" -File |
    Sort-Object Name |
    ForEach-Object {
        "=== Profilstand $($_.BaseName) | $($_.LastWriteTime.ToString('yyyy-MM-dd HH:mm:ss')) ==="

        $fileStream = [System.IO.File]::OpenRead($_.FullName)
        $gzipStream = [System.IO.Compression.GZipStream]::new(
            $fileStream,
            [System.IO.Compression.CompressionMode]::Decompress
        )
        $reader = [System.IO.StreamReader]::new($gzipStream)

        try {
            [xml]$profile = $reader.ReadToEnd()
            $profile.profile.units.unit |
                Where-Object {
                    $_.id -like "org.objectweb.asm*" -or
                    ($_.id -like "com.sap.adt*" -and $_.id -like "*.feature.group")
                } |
                Sort-Object id, version |
                ForEach-Object { "$($_.id) | $($_.version)" }
        }
        finally {
            $reader.Dispose()
            $gzipStream.Dispose()
            $fileStream.Dispose()
        }
    }
